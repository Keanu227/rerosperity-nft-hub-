        fetch(scriptsFolder)
            .then(response => response.text())
            .then(data => {
                // Not all servers list directories, fallback to manual list
                const scriptFiles = ['newfile.js']; // add any new files here initially
                scriptFiles.forEach(file => {
                    const script = document.createElement('script');
                    script.src = `${scriptsFolder}${file}`;
                    document.body.appendChild(script);
                });
            });
    </script>
